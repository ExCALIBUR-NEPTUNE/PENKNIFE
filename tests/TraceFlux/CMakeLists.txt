cmake_minimum_required(VERSION 3.21)
project(PENKNIFETraceFluxChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(NESO REQUIRED)
find_package(MPI REQUIRED)
enable_testing()
get_filename_component(PENKNIFE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(TRACE_FACTORY "${PENKNIFE_ROOT}/src/FluxTraceAdvDiv/FluxTraceAdvDiffFactory.cpp")

add_library(trace_factory OBJECT ${TRACE_FACTORY})
target_include_directories(trace_factory PRIVATE "${PENKNIFE_ROOT}/src")
target_link_libraries(trace_factory PRIVATE Nektar++::nektar++)
foreach(check IN ITEMS Kernels Operator Diffusion)
  add_executable(check_trace_${check} Check${check}.cpp $<TARGET_OBJECTS:trace_factory>)
  target_include_directories(check_trace_${check} PRIVATE "${PENKNIFE_ROOT}/src")
  target_link_libraries(check_trace_${check} PRIVATE Nektar++::nektar++)
endforeach()
add_test(NAME kernels COMMAND check_trace_Kernels)
add_test(NAME periodic COMMAND check_trace_Operator "${CMAKE_CURRENT_SOURCE_DIR}/periodic.xml")
add_test(NAME dirichlet COMMAND check_trace_Operator "${CMAKE_CURRENT_SOURCE_DIR}/dirichlet.xml")
if(MPIEXEC_EXECUTABLE)
  add_test(NAME periodic_mpi2 COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 2
           ${MPIEXEC_PREFLAGS} $<TARGET_FILE:check_trace_Operator>
           "${CMAKE_CURRENT_SOURCE_DIR}/periodic.xml" ${MPIEXEC_POSTFLAGS})
  set_tests_properties(periodic_mpi2 PROPERTIES PROCESSORS 2)
endif()
set_tests_properties(kernels periodic dirichlet PROPERTIES TIMEOUT 90)

target_include_directories(check_trace_Diffusion PRIVATE "${PENKNIFE_ROOT}/NESO/nektar/solvers")
add_test(NAME anisotropic_diffusion COMMAND check_trace_Diffusion)
set_tests_properties(anisotropic_diffusion PROPERTIES TIMEOUT 90)

option(PENKNIFE_TRACE_TEST_AVX2 "Also check fused diffusion with AVX2 (requires an AVX2 CPU)" OFF)
if(PENKNIFE_TRACE_TEST_AVX2)
  target_compile_options(check_trace_Diffusion PRIVATE -mavx2)
endif()
