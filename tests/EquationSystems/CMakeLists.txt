add_executable(check_equation_systems CheckSystems.cpp CheckParticles.cpp $<TARGET_OBJECTS:ObjLib>)
target_include_directories(check_equation_systems PRIVATE ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(check_equation_systems PRIVATE NESO-Particles::NESO-Particles
  Nektar++::nektar++ NESO::nesolib NESO-RNG-Toolkit::NESO-RNG-Toolkit
  VANTAGE-Reactions::VANTAGE-Reactions)
add_sycl_to_target(TARGET check_equation_systems SOURCES CheckSystems.cpp CheckParticles.cpp)
foreach(MODEL IN ITEMS AnisotropicDiffusion ReducedBraginskii ElectrostaticTurbulence NullSystem)
  set(CLOSURES NoClosure)
  if(MODEL STREQUAL "AnisotropicDiffusion")
    set(CLOSURES SingleDiffusive DoubleDiffusive)
    set(projections DisContinuous Continuous)
  else()
    set(CLOSURES NoClosure)
    set(projections DisContinuous)
  endif()
  foreach(CLOSURE IN LISTS CLOSURES)
    set(CLOSURE_INFO "")
    set(CLOSURE_SUFFIX "")
    if(NOT CLOSURE STREQUAL "NoClosure")
      set(CLOSURE_INFO "<I PROPERTY=\"Closure\" VALUE=\"${CLOSURE}\" />")
      set(CLOSURE_SUFFIX "${CLOSURE}_")
    endif()
  foreach(PROJECTION IN LISTS projections)
    set(SECOND_ION "")
    set(MANUFACTURED 0)
    set(DENSITY 2)
    set(BX 0)
    set(FIELDS "n,v,e")
    if(CLOSURE STREQUAL "SingleDiffusive")
      set(FIELDS n)
    endif()
    set(name "${MODEL}_${CLOSURE_SUFFIX}${PROJECTION}")
    configure_file(conditions.xml.in "${name}.xml" @ONLY)
    add_test(NAME "equations_${name}" COMMAND check_equation_systems "${CMAKE_CURRENT_BINARY_DIR}/${name}.xml")
    set_tests_properties("equations_${name}" PROPERTIES TIMEOUT 90
      ENVIRONMENT "OMP_NUM_THREADS=2;ACPP_VISIBILITY_MASK=omp")
    if(PROJECTION STREQUAL "Continuous" OR MODEL STREQUAL "ReducedBraginskii" OR MODEL STREQUAL "ElectrostaticTurbulence")
      set(SECOND_ION "<I NAME=\"zeta\" FIELDS=\"${FIELDS}\"><P>Mass = 4</P><P>Charge = 2</P></I>")
      configure_file(conditions.xml.in "${name}_multispecies.xml" @ONLY)
      add_test(NAME "equations_${name}_multispecies" COMMAND check_equation_systems "${CMAKE_CURRENT_BINARY_DIR}/${name}_multispecies.xml")
      set_tests_properties("equations_${name}_multispecies" PROPERTIES TIMEOUT 90
        ENVIRONMENT "OMP_NUM_THREADS=2;ACPP_VISIBILITY_MASK=omp")
      set(SECOND_ION "")
    endif()
    if(CLOSURE STREQUAL "SingleDiffusive")
      set(MANUFACTURED 1)
      set(DENSITY "2+0.1*sin(PI*x)")
      set(BX 1)
      configure_file(conditions.xml.in "${name}_manufactured.xml" @ONLY)
      add_test(NAME "equations_${name}_manufactured" COMMAND check_equation_systems "${CMAKE_CURRENT_BINARY_DIR}/${name}_manufactured.xml")
      set_tests_properties("equations_${name}_manufactured" PROPERTIES TIMEOUT 90
        ENVIRONMENT "OMP_NUM_THREADS=2;ACPP_VISIBILITY_MASK=omp")
    endif()
  endforeach()
  endforeach()
endforeach()
