cmake_minimum_required(VERSION 3.21)
project(BohmChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(NESO REQUIRED)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BASE_ROOT "${ROOT}" CACHE PATH "PENKNIFE source root (override for a proposal overlay)")
get_filename_component(PROPOSAL "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(bohm_factory OBJECT ${PROPOSAL}/src/BndCond/BndCondBohm/BndCondBohmFactory.cpp
 ${PROPOSAL}/src/FluxTraceAdvDiv/FluxTraceAdvDiffFactory.cpp)
target_include_directories(bohm_factory BEFORE PRIVATE ${PROPOSAL}/src ${BASE_ROOT}/src)
target_link_libraries(bohm_factory PRIVATE Nektar++::nektar++)
add_executable(check_bohm CheckBohm.cpp $<TARGET_OBJECTS:bohm_factory>)
target_include_directories(check_bohm BEFORE PRIVATE ${PROPOSAL}/src ${BASE_ROOT}/src)
target_link_libraries(check_bohm PRIVATE Nektar++::nektar++)
add_test(NAME bohm COMMAND check_bohm)

add_executable(check_bohm_operator CheckOperator.cpp $<TARGET_OBJECTS:bohm_factory>)
target_include_directories(check_bohm_operator BEFORE PRIVATE ${PROPOSAL}/src ${BASE_ROOT}/src)
target_link_libraries(check_bohm_operator PRIVATE Nektar++::nektar++)
add_test(NAME bohm_operator COMMAND check_bohm_operator ${CMAKE_CURRENT_SOURCE_DIR}/bohm.xml)
foreach(check IN ITEMS Kernels Operator Diffusion)
  add_executable(regression_${check} ${BASE_ROOT}/tests/TraceFlux/Check${check}.cpp $<TARGET_OBJECTS:bohm_factory>)
  target_include_directories(regression_${check} BEFORE PRIVATE ${PROPOSAL}/src ${BASE_ROOT}/src ${BASE_ROOT}/NESO/nektar/solvers)
  target_link_libraries(regression_${check} PRIVATE Nektar++::nektar++)
endforeach()
add_test(NAME trace_kernels COMMAND regression_Kernels)
add_test(NAME trace_diffusion COMMAND regression_Diffusion)
add_test(NAME trace_periodic COMMAND regression_Operator ${BASE_ROOT}/tests/TraceFlux/periodic.xml)
add_test(NAME trace_dirichlet COMMAND regression_Operator ${BASE_ROOT}/tests/TraceFlux/dirichlet.xml)

add_test(NAME bohm_operator_mixed COMMAND check_bohm_operator ${CMAKE_CURRENT_SOURCE_DIR}/mixed.xml)
